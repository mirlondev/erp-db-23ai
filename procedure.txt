CREATE OR REPLACE PACKAGE pkg_doc AS

  -- Créer un document (retourne doc_id)
  FUNCTION create_document(
    p_doc_type_code IN VARCHAR2,
    p_company_code  IN VARCHAR2,
    p_doc_date      IN DATE,
    p_party_code    IN VARCHAR2 DEFAULT NULL,
    p_user_code     IN VARCHAR2
  ) RETURN NUMBER;

  -- Ajouter une ligne
  PROCEDURE add_line(
    p_doc_id        IN NUMBER,
    p_line_no       IN NUMBER,
    p_product_code  IN VARCHAR2,
    p_qty           IN NUMBER,
    p_unit_price    IN NUMBER,
    p_unit_code     IN VARCHAR2 DEFAULT 'PCS'
  );

  -- Calculer les totaux du document
  PROCEDURE compute_totals(p_doc_id IN NUMBER);

  -- Valider un document
  PROCEDURE validate_document(
    p_doc_id    IN NUMBER,
    p_user_code IN VARCHAR2
  );

  -- Annuler un document
  PROCEDURE cancel_document(
    p_doc_id    IN NUMBER,
    p_user_code IN VARCHAR2,
    p_reason    IN VARCHAR2 DEFAULT NULL
  );

  -- Récupérer le total TTC d'un document
  FUNCTION get_total_ttc(p_doc_id IN NUMBER) RETURN NUMBER;

END pkg_doc;

CREATE OR REPLACE PACKAGE BODY pkg_doc AS

  FUNCTION create_document(
    p_doc_type_code IN VARCHAR2,
    p_company_code  IN VARCHAR2,
    p_doc_date      IN DATE,
    p_party_code    IN VARCHAR2 DEFAULT NULL,
    p_user_code     IN VARCHAR2
  ) RETURN NUMBER IS
    v_doc_id    NUMBER;
    v_doc_no    NUMBER;
    v_party_name VARCHAR2(50);
  BEGIN
    -- Numéro suivant
    SELECT NVL(MAX(doc_no), 0) + 1 INTO v_doc_no
      FROM doc_header WHERE doc_type_code = p_doc_type_code;

    -- Nom du tiers
    IF p_party_code IS NOT NULL THEN
      BEGIN
        SELECT party_name INTO v_party_name
          FROM app_party.party WHERE party_code = p_party_code;
      EXCEPTION WHEN NO_DATA_FOUND THEN v_party_name := NULL;
      END;
    END IF;

    INSERT INTO doc_header (
      doc_type_code, doc_no, company_code, doc_date,
      party_code, party_name, status, created_by, created_at
    ) VALUES (
      p_doc_type_code, v_doc_no, p_company_code, p_doc_date,
      p_party_code, v_party_name, 'DRAFT', p_user_code, SYSDATE
    ) RETURNING doc_id INTO v_doc_id;

    RETURN v_doc_id;
  END create_document;

  PROCEDURE add_line(
    p_doc_id        IN NUMBER,
    p_line_no       IN NUMBER,
    p_product_code  IN VARCHAR2,
    p_qty           IN NUMBER,
    p_unit_price    IN NUMBER,
    p_unit_code     IN VARCHAR2 DEFAULT 'PCS'
  ) IS
    v_doc_type  VARCHAR2(20);
    v_wh        VARCHAR2(5);
    v_desc      VARCHAR2(120);
  BEGIN
    SELECT doc_type_code, warehouse_code INTO v_doc_type, v_wh
      FROM doc_header WHERE doc_id = p_doc_id;

    BEGIN
      SELECT product_name INTO v_desc
        FROM app_product.product WHERE product_code = p_product_code;
    EXCEPTION WHEN NO_DATA_FOUND THEN v_desc := p_product_code;
    END;

    INSERT INTO doc_line (
      doc_id, line_no, order_no, doc_type_code, warehouse_code,
      product_code, description_1, quantity, unit_code, unit_price,
      amount, unit_price_ht, amount_ht, line_type, is_line_closed
    ) VALUES (
      p_doc_id, p_line_no, p_line_no, v_doc_type, v_wh,
      p_product_code, v_desc, p_qty, p_unit_code, p_unit_price,
      p_qty * p_unit_price,
      ROUND(p_qty * p_unit_price / 1.189, 4),
      ROUND(p_qty * p_unit_price / 1.189, 4),
      'STD', 'N'
    );
  END add_line;

  PROCEDURE compute_totals(p_doc_id IN NUMBER) IS
  BEGIN
    UPDATE doc_header
       SET total_ht  = (SELECT NVL(SUM(amount_ht), 0) FROM doc_line WHERE doc_id = p_doc_id),
           total_ttc = (SELECT NVL(SUM(amount), 0)    FROM doc_line WHERE doc_id = p_doc_id)
     WHERE doc_id = p_doc_id;
  END compute_totals;

  PROCEDURE validate_document(
    p_doc_id    IN NUMBER,
    p_user_code IN VARCHAR2
  ) IS
  BEGIN
    compute_totals(p_doc_id);

    UPDATE doc_header
       SET status     = 'VALID',
           updated_by = p_user_code,
           updated_at = SYSDATE
     WHERE doc_id = p_doc_id;

    INSERT INTO doc_log (doc_id, operation_at, user_code, operation_label)
    VALUES (p_doc_id, SYSDATE, p_user_code, 'Document validé');
  END validate_document;

  PROCEDURE cancel_document(
    p_doc_id    IN NUMBER,
    p_user_code IN VARCHAR2,
    p_reason    IN VARCHAR2 DEFAULT NULL
  ) IS
  BEGIN
    UPDATE doc_header
       SET status     = 'CANCELLED',
           updated_by = p_user_code,
           updated_at = SYSDATE
     WHERE doc_id = p_doc_id;

    INSERT INTO doc_log (doc_id, operation_at, user_code, operation_label)
    VALUES (p_doc_id, SYSDATE, p_user_code, 
            'Document annulé : ' || NVL(p_reason, '(sans motif)'));
  END cancel_document;

  FUNCTION get_total_ttc(p_doc_id IN NUMBER) RETURN NUMBER IS
    v_total NUMBER(16,4);
  BEGIN
    SELECT NVL(SUM(amount), 0) INTO v_total
      FROM doc_line WHERE doc_id = p_doc_id;
    RETURN v_total;
  END get_total_ttc;

END pkg_doc;

CREATE OR REPLACE PACKAGE BODY pkg_pos_sales AS

  FUNCTION create_return(
    p_original_terminal IN VARCHAR2,
    p_original_ticket   IN NUMBER,
    p_user_code         IN VARCHAR2,
    p_reason            IN VARCHAR2 DEFAULT NULL
  ) RETURN NUMBER IS
    v_return_id     NUMBER;
    v_return_number VARCHAR2(20);
    v_customer      VARCHAR2(32);
  BEGIN
    -- Récupérer le client du ticket original
    BEGIN
      SELECT customer_code INTO v_customer
        FROM ticket WHERE terminal_id = p_original_terminal AND ticket_no = p_original_ticket;
    EXCEPTION WHEN NO_DATA_FOUND THEN v_customer := NULL;
    END;

    v_return_number := 'RET-' || TO_CHAR(SYSDATE,'YYYYMMDD') || '-' || 
                       LPAD(NVL((SELECT COUNT(*)+1 FROM ticket_return WHERE TRUNC(return_date) = TRUNC(SYSDATE)), 1), 4, '0');

    INSERT INTO ticket_return (
      return_number, return_date, original_terminal, original_ticket_no,
      customer_code, total_return_ttc, total_return_ht, reason, status, processed_by
    ) VALUES (
      v_return_number, SYSDATE, p_original_terminal, p_original_ticket,
      v_customer, 0, 0, p_reason, 'VALID', p_user_code
    ) RETURNING return_id INTO v_return_id;

    RETURN v_return_id;
  END create_return;

  PROCEDURE add_return_line(
    p_return_id       IN NUMBER,
    p_line_no         IN NUMBER,
    p_product_code    IN VARCHAR2,
    p_quantity        IN NUMBER,
    p_unit_price      IN NUMBER,
    p_source_line_no  IN NUMBER DEFAULT NULL
  ) IS
    v_desc VARCHAR2(120);
    v_amount NUMBER(16,4);
  BEGIN
    BEGIN
      SELECT product_name INTO v_desc FROM app_product.product WHERE product_code = p_product_code;
    EXCEPTION WHEN NO_DATA_FOUND THEN v_desc := p_product_code;
    END;

    v_amount := p_quantity * p_unit_price;

    INSERT INTO ticket_return_line (
      return_id, line_no, product_code, description, quantity, unit_price, amount, source_line_no
    ) VALUES (
      p_return_id, p_line_no, p_product_code, v_desc, p_quantity, p_unit_price, v_amount, p_source_line_no
    );

    UPDATE ticket_return
       SET total_return_ttc = total_return_ttc + v_amount,
           total_return_ht  = total_return_ht  + ROUND(v_amount / 1.189, 4)
     WHERE return_id = p_return_id;
  END add_return_line;

  PROCEDURE apply_line_discount(
    p_terminal_id     IN VARCHAR2,
    p_ticket_no       IN NUMBER,
    p_line_no         IN NUMBER,
    p_discount_rate   IN NUMBER,
    p_authorized_by   IN VARCHAR2
  ) IS
    v_amount NUMBER(16,4);
  BEGIN
    SELECT amount INTO v_amount FROM ticket_line 
     WHERE terminal_id = p_terminal_id AND ticket_no = p_ticket_no AND line_no = p_line_no;

    INSERT INTO ticket_discount (
      terminal_id, ticket_no, line_no, discount_type, discount_rate, discount_amount, authorized_by
    ) VALUES (
      p_terminal_id, p_ticket_no, p_line_no, 'LINE', p_discount_rate,
      ROUND(v_amount * p_discount_rate / 100, 4), p_authorized_by
    );

    UPDATE ticket_line
       SET discount_rate = p_discount_rate
     WHERE terminal_id = p_terminal_id AND ticket_no = p_ticket_no AND line_no = p_line_no;
  END apply_line_discount;

  PROCEDURE add_loyalty_points(
    p_terminal_id     IN VARCHAR2,
    p_ticket_no       IN NUMBER,
    p_card_id         IN NUMBER,
    p_points_earned   IN NUMBER
  ) IS
  BEGIN
    INSERT INTO ticket_loyalty (terminal_id, ticket_no, card_id, points_earned, points_used)
    VALUES (p_terminal_id, p_ticket_no, p_card_id, p_points_earned, 0);

    -- Mettre à jour le solde de la carte
    UPDATE app_party.loyalty_card
       SET points_balance = points_balance + p_points_earned,
           last_used_at = SYSDATE
     WHERE card_id = p_card_id;
  END add_loyalty_points;

  PROCEDURE close_cash_session(
    p_terminal_code   IN VARCHAR2,
    p_session_no      IN NUMBER,
    p_actual_cash     IN NUMBER,
    p_user_code       IN VARCHAR2
  ) IS
    v_theo   NUMBER(16,4);
    v_sales  NUMBER(16,4);
    v_nb     NUMBER(8);
  BEGIN
    SELECT NVL(SUM(MTPERCU), 0), COUNT(*) INTO v_theo, v_nb
      FROM ticket WHERE terminal_id = p_terminal_code AND session_no = p_session_no
       AND status = 'V';

    INSERT INTO pos_cash_closure (
      terminal_code, session_no, closure_date,
      theoretical_cash, actual_cash, difference,
      total_sales, nb_tickets, closed_by
    ) VALUES (
      p_terminal_code, p_session_no, SYSDATE,
      v_theo, p_actual_cash, p_actual_cash - v_theo,
      v_theo, v_nb, p_user_code
    );

    -- Clôturer la session
    UPDATE app_pos.pos_session
       SET closed_at = SYSDATE, closing_balance = p_actual_cash
     WHERE terminal_id = (SELECT terminal_id FROM app_pos.pos_terminal WHERE terminal_code = p_terminal_code)
       AND session_no = p_session_no;
  END close_cash_session;

  FUNCTION session_total_sales(
    p_terminal_code   IN VARCHAR2,
    p_session_no      IN NUMBER
  ) RETURN NUMBER IS
    v_total NUMBER(16,4);
  BEGIN
    SELECT NVL(SUM(total_ttc), 0) INTO v_total
      FROM ticket
     WHERE terminal_id = p_terminal_code AND session_no = p_session_no AND status = 'V';
    RETURN v_total;
  END session_total_sales;

END pkg_pos_sales;

PROMPT [6] MV_HOURLY_SALES
CREATE MATERIALIZED VIEW mv_hourly_sales
  TABLESPACE USERS
  BUILD IMMEDIATE
  REFRESH COMPLETE ON DEMAND
AS
SELECT TRUNC(ticket_date, 'HH24') AS sale_hour,
       terminal_id,
       COUNT(*)                   AS ticket_count,
       SUM(total_ht)              AS total_ht,
       SUM(total_ttc)             AS total_ttc,
       SUM(total_ttc - total_ht)  AS total_tax
  FROM app_sales.ticket
 WHERE status = 'V'
 GROUP BY TRUNC(ticket_date, 'HH24'), terminal_id;

CREATE INDEX ix_mv_hourly_sales ON mv_hourly_sales(sale_hour, terminal_id);

-- [2] MV : top articles du mois
PROMPT [7] MV_MONTHLY_TOP_PRODUCTS
CREATE MATERIALIZED VIEW mv_monthly_top_products
  TABLESPACE USERS
  BUILD IMMEDIATE
  REFRESH COMPLETE ON DEMAND
AS
SELECT TRUNC(t.ticket_date, 'MM') AS sale_month,
       l.product_code,
       MAX(p.product_name)        AS product_name,
       SUM(l.quantity)            AS total_qty,
       SUM(l.quantity * l.unit_price) AS total_amount,
       COUNT(DISTINCT t.ticket_no) AS ticket_count
  FROM app_sales.ticket t
  JOIN app_sales.ticket_line l
    ON l.terminal_id = t.terminal_id AND l.ticket_no = t.ticket_no
  LEFT JOIN app_product.product p ON p.product_code = l.product_code
 WHERE t.status = 'V'
 GROUP BY TRUNC(t.ticket_date, 'MM'), l.product_code;

CREATE INDEX ix_mv_monthly_top ON mv_monthly_top_products(sale_month, total_amount DESC);

CREATE OR REPLACE VIEW v_kpi_today AS
SELECT TRUNC(SYSDATE)                             AS report_date,
       COUNT(*)                                   AS tickets_count,
       SUM(total_ttc)                             AS total_revenue,
       SUM(total_ht)                              AS total_ht,
       SUM(total_ttc - total_ht)                  AS total_tax,
       AVG(total_ttc)                             AS avg_ticket,
       COUNT(DISTINCT customer_code)              AS unique_customers
  FROM app_sales.ticket
 WHERE TRUNC(ticket_date) = TRUNC(SYSDATE)
   AND status = 'V';

-- [2] État du stock critique
CREATE OR REPLACE VIEW v_critical_stock AS
SELECT s.warehouse_code,
       s.product_code,
       p.product_name,
       s.stock_qty,
       p.min_stock_qty,
       p.safety_stock_qty,
       CASE 
         WHEN s.stock_qty <= 0 THEN 'RUPTURE'
         WHEN s.stock_qty <= NVL(p.safety_stock_qty, 0) THEN 'CRITIQUE'
         WHEN s.stock_qty <= NVL(p.min_stock_qty, 0) THEN 'ALERTE'
         ELSE 'OK'
       END AS stock_status
  FROM app_inv.inv_stock s
  JOIN app_product.product p ON p.product_code = s.product_code
 WHERE s.stock_qty <= NVL(p.min_stock_qty, 0) * 1.2;

-- [3] Encours clients
CREATE OR REPLACE VIEW v_customer_balance AS
SELECT p.party_code,
       p.party_name,
       p.phone,
       p.credit_limit,
       NVL((SELECT SUM(balance) FROM app_ar.invoice i WHERE i.party_code = p.party_code AND i.status IN ('VALID','PARTIAL')), 0) AS outstanding,
       NVL((SELECT SUM(total_ttc) FROM app_sales.ticket t WHERE t.customer_code = p.party_code AND t.status = 'V'), 0) AS total_purchases,
       CASE 
         WHEN p.credit_limit > 0 THEN 
           ROUND(NVL((SELECT SUM(balance) FROM app_ar.invoice i WHERE i.party_code = p.party_code AND i.status IN ('VALID','PARTIAL')), 0) / p.credit_limit * 100, 2)
         ELSE 0
       END AS utilization_pct
  FROM app_party.party p
 WHERE p.status = 'ACTIVE';
